Asisten AI
Kamu
Tampilkan user dengan order terbanyak bulan ini
AI
SELECT 
  u.name,
  u.email,
  COUNT(o.id) AS order_count
FROM users u
JOIN orders o ON o.user_id = u.id
WHERE o.created_at >= DATE_TRUNC('month', NOW())
GROUP BY u.id, u.name, u.email
ORDER BY order_count DESC
LIMIT 10;
